cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DexFragglerNativeReference VERSION 1.0.0 LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(GLOB DX7_MSFA CONFIGURE_DEPENDS vendor/third_party/dx7/msfa/*.cc)
add_library(dexfraggler_dx7_engine STATIC
    vendor/Source/Patch.cpp vendor/Source/Dx7Engine.cpp
    vendor/third_party/dx7/EngineMkI.cpp
    vendor/third_party/dx7/msfa/porta.cpp ${DX7_MSFA})
add_library(dx7_reference ALIAS dexfraggler_dx7_engine)
set_target_properties(dexfraggler_dx7_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(dexfraggler_dx7_engine PUBLIC vendor/Source PRIVATE vendor/third_party/dx7)
target_compile_definitions(dexfraggler_dx7_engine PUBLIC _USE_MATH_DEFINES NOMINMAX)
if(MSVC)
    target_compile_options(dexfraggler_dx7_engine PRIVATE /O2 /W3 /wd4244 /wd4267 /wd4996)
endif()
add_executable(DexfragglerReference NativeRenderer.cpp)
target_link_libraries(DexfragglerReference PRIVATE dexfraggler_dx7_engine)
add_executable(WaveFinderEngineTests tests/WaveFinderEngineTests.cpp)
target_link_libraries(WaveFinderEngineTests PRIVATE dexfraggler_dx7_engine)
enable_testing()
add_test(NAME copied_engine_suite COMMAND WaveFinderEngineTests)
